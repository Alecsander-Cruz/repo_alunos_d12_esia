def cpf_valido(cpf):
    d = [int(c) for c in cpf if c.isdigit()]
    if len(d) != 11:
        return False
    for n in (9, 10):
        soma = sum(d[i] * (n + 1 - i) for i in range(n))
        if (soma * 10) % 11 % 10 != d[n]:
            return False
    return True
